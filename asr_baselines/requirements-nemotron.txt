nemo_toolkit[asr]>=2.0

# nemo_toolkit does NOT pull in torchaudio -- install matching the installed
# torch build explicitly, e.g.:
#   pip install torchaudio --index-url https://download.pytorch.org/whl/cu130
# (match the cuXXX suffix to whatever torch version nemo_toolkit pulled in --
# check with `python -c "import torch; print(torch.__version__)"` first).
#
# Recent torchaudio (>=2.9ish) moved load/save entirely onto torchcodec, which
# needs REAL FFmpeg shared libraries (libavutil.so etc), not just a pip
# package -- torchaudio.load()/.save() raise ModuleNotFoundError otherwise.
# pip can't provide these; install via conda-forge instead:
#   conda install -p <env> -c conda-forge "ffmpeg<8"
#   pip install torchcodec --index-url https://download.pytorch.org/whl/cu130
#
# torchaudio.info() no longer exists on this stack -- asr_baselines code
# already falls back to a full torchaudio.load() decode when it's missing
# (see make_nemo_manifest.py), so this is a known, handled compatibility gap,
# not a bug to fix here.
#
# numba: NOT pinned by nemo_toolkit itself (pulled in transitively via
# librosa/resampy), so pip resolves the latest release with no vetting against
# NeMo's numba-CUDA-JIT'd RNNT loss kernels. The unpinned latest (0.66.0 as of
# writing) crashed with a numba internal type-inference error mid-training
# ("Signature mismatch: N argument types given, but function takes M
# arguments"). Pin explicitly to a more established release -- install AFTER
# nemo_toolkit so this downgrade isn't immediately overwritten by its own
# resolution:
#   pip install --no-user "numba==0.60.0"
numba==0.60.0
#
# numba-cuda: since numba 0.59, CUDA target support lives in this SEPARATE
# package (kept up to date with new CUDA releases independently of numba
# itself) -- NOT installed by numba or nemo_toolkit automatically. Without it,
# numba.cuda's real-GPU code path segfaults hard (no Python exception) on a
# very recent CUDA build (confirmed: identical segfault survived 5 unrelated
# fix attempts -- RNNT loss backend, precision/AMP, CUDA graphs, dataloader
# workers, non-fused AdamW -- and was only diagnosed by forcing
# NUMBA_ENABLE_CUDASIM=1, which routes around real CUDA and turned the segfault
# into a clean, catchable AttributeError instead). Install explicitly:
numba-cuda
